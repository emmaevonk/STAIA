    Column name written to adata.obs with the predicted cell types.

def annotate_celltypist(
    adata: ad.AnnData,
    model: str = "Immune_All_Low.pkl",
    majority_voting: bool = True,
    over_clustering: Optional[str] = None,
    result_key: str = "celltypist_cell_type",
    inplace: bool = True
):
    """
    Annotate cells using CellTypist (marker/classifier-based).

    Does *not* require an external scRNA-seq reference — uses pre-trained
    models that ship with the package or can be downloaded automatically.

    Parameters
    ----------
    adata : AnnData
        Spatial or single-cell AnnData object.
        ``adata.X`` should contain normalised, log1p-transformed counts
        (CellTypist normalises to 10,000 counts/cell internally if needed).
    model : str
        CellTypist model name or path to a local ``.pkl`` model file.
        Run ``celltypist.models.get_all_models()`` to list available models.
        Commonly used models:
          - ``"Immune_All_Low.pkl"``   – immune cells, fine-grained
          - ``"Immune_All_High.pkl"``  – immune cells, coarse
          - ``"Adult_Mouse_Gut.pkl"``  – mouse gut atlas
    majority_voting : bool
        Whether to refine predictions by majority voting within clusters.
        Recommended for spatial data. Requires a clustering column to be set
        in ``adata.obs`` (see ``over_clustering``).
    over_clustering : str, optional
        Column in ``adata.obs`` with cluster labels used for majority voting.
        If None, CellTypist performs its own over-clustering automatically.
    result_key : str
        Column name written to ``adata.obs`` with the predicted cell types.

    Returns
    -------
    AnnData or None

    Examples
    --------
    >>> STAIA.annotate_celltypist(adata)
    >>> adata.obs["celltypist_cell_type"]
    """
    import celltypist
    from celltypist import models

    import scipy.sparse as sp
    import scanpy as sc

    X = adata.X
    mat = X.toarray() if sp.issparse(X) else np.asarray(X)

    # Heuristic 1 – raw counts: values are non-negative integers
    looks_raw = (mat >= 0).all() and np.allclose(mat, mat.astype(int))

    # Heuristic 2 – already normalised but not log1p: per-cell sums near 10 000
    #   (log1p data has per-cell sums far below 10 000)
    per_cell_sums = mat.sum(axis=1)
    looks_normalised_not_logged = (per_cell_sums.mean() > 100)

    # Heuristic 3 – already log1p: max value is small (log1p(10000) ≈ 9.2)
    looks_log1p = (mat.max() < 20) and (per_cell_sums.mean() < 100)

    if looks_log1p:
        # Data already looks correctly preprocessed — proceed as-is
        pass
    else:
        if looks_raw:
            warnings.warn(
                "adata.X appears to contain raw counts (non-negative integers). "
                "Automatically applying sc.pp.normalize_total(target_sum=1e4) "
                "and sc.pp.log1p(). The result is stored in "
                "adata.layers['log1p_norm'] and used for CellTypist; "
                "adata.X is left unchanged.",
                UserWarning,
                stacklevel=2,
            )
        else:
            warnings.warn(
                "adata.X does not look like log1p-normalised data "
                f"(mean per-cell sum = {per_cell_sums.mean():.1f}, "
                f"max value = {mat.max():.2f}). "
                "Automatically applying sc.pp.normalize_total(target_sum=1e4) "
                "and sc.pp.log1p(). The result is stored in "
                "adata.layers['log1p_norm'] and used for CellTypist; "
                "adata.X is left unchanged.",
                UserWarning,
                stacklevel=2,
            )

        # Normalise into a temporary copy so we never overwrite .X
        adata_norm = adata.copy()
        sc.pp.normalize_total(adata_norm, target_sum=1e4)
        sc.pp.log1p(adata_norm)

        # Cache the preprocessed matrix as a layer for transparency
        adata.layers["log1p_norm"] = adata_norm.X

        # Point .X at the normalised matrix for the CellTypist call,
        # then restore it afterwards
        _original_X = adata.X
        adata.X = adata_norm.X


    # download model if it is not already cached
    models.download_models(model=model, force_update=False)
    ct_model = models.Model.load(model=model)

    predictions = celltypist.annotate(
        adata,
        model=ct_model,
        majority_voting=majority_voting,
        over_clustering=over_clustering,
    )

    # adata.obs[result_key] = predictions.predicted_labels.predicted_labels.values
    if majority_voting:
        adata.obs[result_key] = predictions.predicted_labels.majority_voting.values
    else:
        adata.obs[result_key] = predictions.predicted_labels.predicted_labels.values
    adata.obs[f"{result_key}_score"] = predictions.probability_matrix.max(axis=1).values

    # store full probability matrix in obsm
    adata.obsm[f"{result_key}_prob_matrix"] = predictions.probability_matrix #TODO: maybe discard later?
    return adata

Annotate spatial data via Tangram optimal-transport cell-to-space mapping.

Requires a scRNA-seq reference with existing cell type labels. Tangram maps single cells onto spatial coordinates and transfers cell-type labels, producing per-spot cell type proportion estimates.

Parameters:
  • adata_sp (AnnData) –

    Spatial AnnData object. Must have obsm["spatial"] with coordinates.

  • adata_sc (AnnData) –

    scRNA-seq reference AnnData with cell type labels in .obs.

  • annotation_key (str) –

    Column in adata_sc.obs containing cell type labels.

  • result_key (str, default: 'tangram_cell_type' ) –

    Prefix for columns written to adata_sp.obsm with cell type fractions (adata_sp.obsm[result_key]) and to adata_sp.obs with the most abundant cell type per spot (adata_sp.obs[result_key + "_dominant"]).

  • mode (str, default: 'clusters' ) –

    Tangram mapping mode: - "clusters" – maps cell-type clusters (faster, recommended) - "cells" – maps individual cells (more granular, slower)

  • num_epochs (int, default: 1000 ) –

    Number of optimisation epochs.

  • device (str, default: 'cpu' ) –

    "cpu" or "cuda" (if a GPU is available).
Returns:
  • AnnData or None –

Examples:

>>> STAIA.annotate_tangram(adata_sp, adata_sc, annotation_key="cell_type")

def annotate_tangram(
    adata_sp: ad.AnnData,
    adata_sc: ad.AnnData,
    annotation_key: str,
    result_key: str = "tangram_cell_type",
    mode: str = "clusters",
    num_epochs: int = 1000,
    device: str = "cpu",
) -> Optional[ad.AnnData]:
    """
    Annotate spatial data via Tangram optimal-transport cell-to-space mapping.

    Requires a scRNA-seq reference with existing cell type labels.
    Tangram maps single cells onto spatial coordinates and transfers
    cell-type labels, producing per-spot cell type proportion estimates.

    Parameters
    ----------
    adata_sp : AnnData
        Spatial AnnData object. Must have ``obsm["spatial"]`` with coordinates.
    adata_sc : AnnData
        scRNA-seq reference AnnData with cell type labels in ``.obs``.
    annotation_key : str
        Column in ``adata_sc.obs`` containing cell type labels.
    result_key : str
        Prefix for columns written to ``adata_sp.obsm`` with cell type
        fractions (``adata_sp.obsm[result_key]``) and to ``adata_sp.obs``
        with the most abundant cell type per spot
        (``adata_sp.obs[result_key + "_dominant"]``).
    mode : str
        Tangram mapping mode:
          - ``"clusters"`` – maps cell-type clusters (faster, recommended)
          - ``"cells"``    – maps individual cells (more granular, slower)
    num_epochs : int
        Number of optimisation epochs.
    device : str
        ``"cpu"`` or ``"cuda"`` (if a GPU is available).

    Returns
    -------
    AnnData or None

    Examples
    --------
    >>> STAIA.annotate_tangram(adata_sp, adata_sc, annotation_key="cell_type")
    >>> adata_sp.obs["tangram_cell_type_dominant"]
    """
    import tangram as tg

    # find shared markers between scRNAseq and spatial dataset
    tg.pp_adatas(adata_sc, adata_sp, genes=None)

    # learn cell-to-space mapping
    ad_map = tg.map_cells_to_space(
        adata_sc,
        adata_sp,
        mode=mode,
        cluster_label=annotation_key,
        num_epochs=num_epochs,
        device=device,
        verbose=False,
    )

    # project cell type annotations onto spatial spots
    tg.project_cell_annotations(
        ad_map,
        adata_sp,
        annotation=annotation_key,
    )

    # store fractions in obsm and dominant type in obs
    ct_df: pd.DataFrame = adata_sp.obsm["tangram_ct_pred"]
    adata_sp.obsm[result_key] = ct_df
    adata_sp.obs[f"{result_key}_dominant"] = ct_df.idxmax(axis=1).values
    return adata_sp

annotate_consensus(adata, obsm_keys=None, obs_keys=None, weights=None, strategy='weighted_average', result_key='consensus', min_methods=2, auto_detect=True)

Build a consensus cell type annotation from multiple method outputs.

Reads per-spot cell type information already stored in adata.obsm (fraction/probability matrices) and/or adata.obs (dominant-label columns), and combines them into a single consensus call.

Parameters:
  • adata (AnnData) –

    Spatial AnnData that has been annotated by one or more methods.

  • obsm_keys (dict, default: None ) –

    Mapping of {method_name: obsm_key} pointing to a (spots × cell_types) fraction DataFrame for each method. Example::

    obsm_keys = {
    "celltypist": "celltypist_cell_type_prob_matrix",
    "my_method":  "my_method_fractions",
}

    If None and auto_detect=True, all known and unknown fraction DataFrames in adata.obsm are used automatically.

  • obs_keys (dict, default: None ) –

    Mapping of {method_name: obs_column} for methods that only stored a single dominant label per spot in adata.obs. Example::

    obs_keys = {
    "sctype":  "sctype_cell_type",
    "manual":  "manual_annotation",
}

    These columns are one-hot encoded internally so they participate in all three consensus strategies on equal footing with fraction matrices. If None and auto_detect=True, obs columns ending in _cell_type or _dominant are picked up automatically.

  • weights (dict, default: None ) –

    Per-method weights for the "weighted_average" strategy. Keys must match those in obsm_keys / obs_keys. Missing keys default to 1.0. Weights are normalised internally. Ignored by "majority_vote" and "rank_aggregation".

  • strategy (('weighted_average', 'majority_vote', 'rank_aggregation'), default: "weighted_average" ) –

    How to combine method outputs:

    • "weighted_average" – weighted mean of fraction matrices. Best when methods produce calibrated probability outputs.
    • "majority_vote" – plurality vote on each method's dominant label. Best when you trust per-method calls over continuous scores.
    • "rank_aggregation" – Borda-count: ranks cell types per method per spot, then sums ranks. Robust to scale differences between methods.
  • result_key (str, default: 'consensus' ) –

    Prefix for output keys written to adata:

    • adata.obsm[f"{result_key}_fractions"]
    • adata.obs[f"{result_key}_cell_type"]
    • adata.obs[f"{result_key}_confidence"]
    • adata.obs[f"{result_key}_agreement"]
    • adata.uns[f"{result_key}_methods"]
    • adata.uns[f"{result_key}_strategy"]
  • min_methods (int, default: 2 ) –

    Minimum number of methods that must be found before attempting consensus. Raises ValueError if fewer are available.

  • auto_detect (bool, default: True ) –

    When obsm_keys and obs_keys are both None, automatically scan adata.obsm and adata.obs for annotation results. Set to False to require explicit keys.
Returns:
  • AnnData –

    The input adata with consensus results written in place.
Raises:
  • ValueError –

    If fewer than min_methods sources are found.

Examples:

Auto-detect everything (after running celltypist + tangram + tacco):

>>> STAIA.annotate_consensus(adata)
>>> adata.obs["consensus_cell_type"]

Use only specific obsm fraction matrices:

>>> STAIA.annotate_consensus(
...     adata,
...     obsm_keys={
...         "celltypist": "celltypist_cell_type_prob_matrix",
...         "tangram":    "tangram_cell_type",
...     },
... )

Mix fraction matrices with a custom obs column:

>>> STAIA.annotate_consensus(
...     adata,
...     obsm_keys={"tacco": "tacco_cell_type"},
...     obs_keys={"sctype": "sctype_cell_type"},
... )

Weight celltypist more heavily:

>>> STAIA.annotate_consensus(
...     adata,
...     weights={"celltypist": 2.0, "tangram": 1.0, "tacco": 1.0},
... )

Use majority vote with a custom result key:

>>> STAIA.annotate_consensus(adata, strategy="majority_vote", result_key="vote")
>>> adata.obs["vote_cell_type"]

        These columns are one-hot encoded internally so they participate
        in all three consensus strategies on equal footing with fraction
        matrices. If None and ``auto_detect=True``, obs columns ending
        in ``_cell_type`` or ``_dominant`` are picked up automatically.
    weights : dict, optional
        Per-method weights for the ``"weighted_average"`` strategy.
        Keys must match those in ``obsm_keys`` / ``obs_keys``. Missing
        keys default to 1.0. Weights are normalised internally.
        Ignored by ``"majority_vote"`` and ``"rank_aggregation"``.
    strategy : {"weighted_average", "majority_vote", "rank_aggregation"}
        How to combine method outputs:

        - ``"weighted_average"``  – weighted mean of fraction matrices.
          Best when methods produce calibrated probability outputs.
        - ``"majority_vote"``     – plurality vote on each method's dominant
          label. Best when you trust per-method calls over continuous scores.
        - ``"rank_aggregation"``  – Borda-count: ranks cell types per method
          per spot, then sums ranks. Robust to scale differences between
          methods.
    result_key : str
        Prefix for output keys written to ``adata``:

        - ``adata.obsm[f"{result_key}_fractions"]``
        - ``adata.obs[f"{result_key}_cell_type"]``
        - ``adata.obs[f"{result_key}_confidence"]``
        - ``adata.obs[f"{result_key}_agreement"]``
        - ``adata.uns[f"{result_key}_methods"]``
        - ``adata.uns[f"{result_key}_strategy"]``

    min_methods : int
        Minimum number of methods that must be found before attempting
        consensus. Raises ``ValueError`` if fewer are available.
    auto_detect : bool
        When ``obsm_keys`` and ``obs_keys`` are both None, automatically
        scan ``adata.obsm`` and ``adata.obs`` for annotation results.
        Set to False to require explicit keys.

    Returns
    -------
    AnnData
        The input ``adata`` with consensus results written in place.

    Raises
    ------
    ValueError
        If fewer than ``min_methods`` sources are found.

    Examples
    --------
    **Auto-detect everything** (after running celltypist + tangram + tacco):

    >>> STAIA.annotate_consensus(adata)
    >>> adata.obs["consensus_cell_type"]

    **Use only specific obsm fraction matrices:**

    >>> STAIA.annotate_consensus(
    ...     adata,
    ...     obsm_keys={
    ...         "celltypist": "celltypist_cell_type_prob_matrix",
    ...         "tangram":    "tangram_cell_type",
    ...     },
    ... )

    **Mix fraction matrices with a custom obs column:**

    >>> STAIA.annotate_consensus(
    ...     adata,
    ...     obsm_keys={"tacco": "tacco_cell_type"},
    ...     obs_keys={"sctype": "sctype_cell_type"},
    ... )

    **Weight celltypist more heavily:**

    >>> STAIA.annotate_consensus(
    ...     adata,
    ...     weights={"celltypist": 2.0, "tangram": 1.0, "tacco": 1.0},
    ... )

    **Use majority vote with a custom result key:**

    >>> STAIA.annotate_consensus(adata, strategy="majority_vote", result_key="vote")
    >>> adata.obs["vote_cell_type"]
    """
    weights = weights or {}

    # 1. Gather all fraction matrices
    available, skipped = _collect_fraction_matrices(
        adata, obsm_keys, obs_keys, auto_detect
    )

    if skipped:
        warnings.warn(
            "The following sources were not found and will be skipped:\n  "
            + "\n  ".join(skipped)
            + "\nRun the corresponding annotation methods first, or check "
            "your key names.",
            UserWarning,
            stacklevel=2,
        )

    if len(available) < min_methods:
        raise ValueError(
            f"Only {len(available)} annotation source(s) found "
            f"({list(available)}), but min_methods={min_methods}. "
            f"Run at least {min_methods} annotation methods before calling "
            f"annotate_consensus(), or lower min_methods."
        )

    print(
        f"Building consensus from {len(available)} method(s) "
        f"using strategy='{strategy}': {list(available)}"
    )

    # 2. Align cell type columns across all methods (union, fill missing with 0)
    all_cell_types: List[str] = sorted(
        set().union(*[set(df.columns) for df in available.values()])
    )
    aligned: Dict[str, pd.DataFrame] = {
        method: df.reindex(columns=all_cell_types, fill_value=0.0)
        for method, df in available.items()
    }

    # 3. Build consensus fractions
    if strategy == "weighted_average":
        consensus_fractions = _weighted_average(aligned, weights)
    elif strategy == "majority_vote":
        consensus_fractions = _majority_vote(aligned, all_cell_types)
    elif strategy == "rank_aggregation":
        consensus_fractions = _rank_aggregation(aligned)
    else:
        raise ValueError(
            f"Unknown strategy {strategy!r}. "
            f"Choose from 'weighted_average', 'majority_vote', 'rank_aggregation'."
        )

    # Row-normalise (guard against zero-sum rows)
    row_sums = consensus_fractions.sum(axis=1).replace(0, 1)
    consensus_fractions = consensus_fractions.div(row_sums, axis=0)

    # 4. Derive per-spot summary statistics
    dominant_cell_type = consensus_fractions.idxmax(axis=1)
    confidence         = consensus_fractions.max(axis=1)

    method_dominant = pd.DataFrame(
        {m: df.idxmax(axis=1) for m, df in aligned.items()},
        index=adata.obs_names,
    )
    agreement = method_dominant.eq(dominant_cell_type, axis=0).mean(axis=1)

    # 5. Write results back to adata
    adata.obsm[f"{result_key}_fractions"]  = consensus_fractions
    adata.obs[f"{result_key}_cell_type"]   = dominant_cell_type.values
    adata.obs[f"{result_key}_confidence"]  = confidence.astype(float).values
    adata.obs[f"{result_key}_agreement"]   = agreement.astype(float).values
    adata.uns[f"{result_key}_methods"]     = list(available.keys())
    adata.uns[f"{result_key}_strategy"]    = strategy

    return adata
